本研究的 C++ 代码为：

\begin{lstlisting}[language=C++, caption={Fibonacci sequence in C++}]
#include <iostream>
using namespace std;

int fibonacci(int n) {
    if (n <= 1)
        return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main() {
    int n = 10;
    for (int i = 0; i < n; i++) {
        cout << fibonacci(i) << " ";
    }
    return 0;
}
\end{lstlisting}
